## `apps/api/app/core/vapid.py`

```python
"""VAPID key management for Web Push (Phase 10).

Keys come from settings (VAPID_PRIVATE_KEY / VAPID_PUBLIC_KEY) when present;
otherwise a P-256 pair is generated once and persisted to a local JSON file so
the demo works out of the box without committed secrets. Both representations
are the base64url "raw" forms understood by py_vapid and by the browser's
PushManager (applicationServerKey).

Run `python -m app.core.vapid` to print a fresh pair for .env files.
"""

import base64
import json
import logging
from pathlib import Path

from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric import ec

from app.core.config import get_settings

logger = logging.getLogger(__name__)


def _b64url(raw: bytes) -> str:
    return base64.urlsafe_b64encode(raw).decode("ascii").rstrip("=")


def generate_keypair() -> tuple[str, str]:
    """Return (private_key, public_key) as base64url raw strings.

    The private key is the 32-byte big-endian private value (the "raw" form
    py_vapid expects); the public key is the 65-byte uncompressed EC point
    the browser's PushManager understands.
    """
    private_key = ec.generate_private_key(ec.SECP256R1())
    private_value = private_key.private_numbers().private_value
    private_raw = private_value.to_bytes(32, byteorder="big")
    public_raw = private_key.public_key().public_bytes(
        encoding=serialization.Encoding.X962,
        format=serialization.PublicFormat.UncompressedPoint,
    )
    return _b64url(private_raw), _b64url(public_raw)


def load_or_create_keypair() -> tuple[str, str]:
    """Resolve VAPID keys: settings first, then the keys file, else generate."""
    settings = get_settings()
    if settings.vapid_private_key and settings.vapid_public_key:
        return settings.vapid_private_key, settings.vapid_public_key

    keys_file = Path(settings.vapid_keys_file)
    if keys_file.exists():
        try:
            data = json.loads(keys_file.read_text(encoding="utf-8"))
            if data.get("private_key") and data.get("public_key"):
                return data["private_key"], data["public_key"]
        except (OSError, json.JSONDecodeError):
            logger.warning("Could not read VAPID keys file %s; regenerating.", keys_file)

    private_key, public_key = generate_keypair()
    try:
        keys_file.write_text(
            json.dumps({"private_key": private_key, "public_key": public_key}, indent=2),
            encoding="utf-8",
        )
        logger.info("Generated new VAPID keypair at %s", keys_file)
    except OSError:
        logger.warning("Could not persist VAPID keys to %s", keys_file)
    return private_key, public_key


def public_key_for_browser() -> str:
    """The base64url P-256 public key handed to PushManager.subscribe()."""
    return load_or_create_keypair()[1]


if __name__ == "__main__":
    private_key, public_key = generate_keypair()
    print(f"VAPID_PRIVATE_KEY={private_key}")
    print(f"VAPID_PUBLIC_KEY={public_key}")

```

## `apps/api/app/services/push_service.py`

```python
"""Web Push delivery (Phase 10): send encrypted notifications to a user's
browsers via their push services, prune dead subscriptions.

Delivery is fire-and-forget: the realtime event publisher snapshots the
subscription rows and hands them here; network I/O runs in a worker thread so
request latency is untouched, and every failure is logged, never raised.
"""

import asyncio
import json
import logging
from datetime import UTC, datetime
from typing import TYPE_CHECKING

from pywebpush import WebPushException, webpush
from sqlalchemy import delete, select

from app.core.config import get_settings
from app.core.db import SessionLocal
from app.core.vapid import load_or_create_keypair
from app.models.push import PushSubscription

if TYPE_CHECKING:
    from sqlalchemy.orm import Session

logger = logging.getLogger(__name__)

PushTarget = tuple[str, str, str]  # (endpoint, p256dh, auth)


def snapshot_subscriptions(db: "Session", user_ids: list) -> list[PushTarget]:
    """Read subscription rows synchronously (inside the caller's session)."""
    if not user_ids:
        return []
    rows = db.execute(
        select(
            PushSubscription.endpoint,
            PushSubscription.p256dh,
            PushSubscription.auth,
        ).where(PushSubscription.user_id.in_(user_ids))
    ).all()
    return [(row[0], row[1], row[2]) for row in rows]


def _send_one(target: PushTarget, payload: dict) -> None:
    """Blocking single send; raises on failure for the caller to classify."""
    private_key, _ = load_or_create_keypair()
    settings = get_settings()
    endpoint, p256dh, auth = target
    webpush(
        subscription_info={"endpoint": endpoint, "keys": {"p256dh": p256dh, "auth": auth}},
        data=json.dumps(payload),
        vapid_private_key=private_key,
        vapid_claims={"sub": settings.push_subject},
    )


def _prune(endpoints: list[str]) -> None:
    with SessionLocal() as session:
        session.execute(delete(PushSubscription).where(PushSubscription.endpoint.in_(endpoints)))
        session.commit()


def _dispatch(targets: list[PushTarget], payload: dict, user_ids: list) -> None:
    """Send to every target in this thread; prune endpoints the push service
    reports as gone (404/410)."""
    sent = 0
    failed = 0
    gone: list[str] = []
    for target in targets:
        try:
            _send_one(target, payload)
            sent += 1
        except WebPushException as error:
            response = getattr(error, "response", None)
            status_code = getattr(response, "status_code", None)
            if status_code in (404, 410):
                gone.append(target[0])
            else:
                failed += 1
                logger.warning("Push to %s failed: %s", target[0][:64], error)
        except Exception as error:
            failed += 1
            logger.warning("Push to %s failed unexpectedly: %s", target[0][:64], error)

    if sent and user_ids:
        with SessionLocal() as session:
            for user_id in user_ids:
                session.execute(
                    PushSubscription.__table__.update()
                    .where(PushSubscription.user_id == user_id)
                    .values(last_pushed_at=datetime.now(UTC))
                )
            session.commit()
    if gone:
        _prune(gone)

    if sent or failed:
        logger.info("Web push batch: sent=%d failed=%d pruned=%d", sent, failed, len(gone))


def dispatch_push_async(targets: list[PushTarget], payload: dict, user_ids: list) -> None:
    """Schedule the blocking sends in a worker thread from async code."""
    if not targets:
        return
    asyncio.get_running_loop().run_in_executor(None, _dispatch, targets, payload, user_ids)


def send_to_user_sync(db: "Session", user_id, payload: dict) -> dict:
    """Synchronous send used by the /push/test endpoint and tests.

    Returns counters; prunes subscriptions that the push service reports as
    gone. Runs in-thread, so tests can monkeypatch ``_send_one``.
    """
    targets = snapshot_subscriptions(db, [user_id])
    sent = 0
    failed = 0
    gone: list[str] = []
    for target in targets:
        try:
            _send_one(target, payload)
            sent += 1
        except WebPushException as error:
            status_code = getattr(getattr(error, "response", None), "status_code", None)
            if status_code in (404, 410):
                gone.append(target[0])
            else:
                failed += 1
        except Exception:
            failed += 1
    if gone:
        db.execute(delete(PushSubscription).where(PushSubscription.endpoint.in_(gone)))
        db.flush()
    return {"sent": sent, "failed": failed, "pruned": len(gone)}

```

## `apps/api/app/services/realtime.py`

```python
"""Event publishing: audit row + notifications + realtime broadcast + web push.

Every mutating endpoint calls :func:`publish_event` so the audit log
(``events`` table), user notifications and the WebSocket audience stay in
sync. Rows are flushed within the caller's request transaction and committed
by ``get_db`` when the request finishes. Every notification row also fans out
to the user's registered browsers as an encrypted Web Push message
(Phase 10); delivery runs in a worker thread and never fails requests.
"""

import logging
from collections.abc import Sequence
from typing import NamedTuple
from uuid import UUID, uuid4

from sqlalchemy.orm import Session

from app.models import Event, Notification
from app.realtime.manager import manager
from app.services import push_service

logger = logging.getLogger(__name__)


class NotificationDraft(NamedTuple):
    user_id: UUID
    title: str
    body: str
    link: str | None = None


async def publish_event(
    db: Session,
    *,
    event_type: str,
    actor_user_id: UUID | None,
    actor_role: str,
    payload: dict | None = None,
    audience_user_ids: Sequence[UUID] = (),
    audience_roles: Sequence[str] = (),
    notifications: Sequence[NotificationDraft] = (),
) -> UUID:
    """Persist an audit event (plus notifications) and broadcast it live.

    Audience channels are ``user:<id>`` and ``role:<ROLE>``; duplicates in
    the audience lists are collapsed while preserving order.
    """
    resolved_payload = dict(payload or {})
    event = Event(
        id=uuid4(),
        type=event_type,
        actor_user_id=actor_user_id,
        actor_role=actor_role,
        payload=resolved_payload,
    )
    db.add(event)
    for draft in notifications:
        db.add(
            Notification(
                id=uuid4(),
                user_id=draft.user_id,
                title=draft.title,
                body=draft.body,
                link=draft.link,
                read=False,
            )
        )
    db.flush()

    message_payload = {**resolved_payload, "event_id": str(event.id)}
    for user_id in dict.fromkeys(audience_user_ids):
        await manager.broadcast(f"user:{user_id}", event_type, message_payload)
    for role in dict.fromkeys(audience_roles):
        await manager.broadcast(f"role:{role}", event_type, message_payload)

    # Web push mirrors the notification rows: one encrypted message per
    # notified user with a browser subscription. Snapshot synchronously here
    # (same transaction), deliver in the background.
    drafts = list(notifications)
    if drafts:
        targets = push_service.snapshot_subscriptions(db, [draft.user_id for draft in drafts])
        if targets:
            first = drafts[0]
            push_payload = {
                "title": first.title,
                "body": first.body,
                "link": first.link or "/dashboard",
                "event_id": str(event.id),
                "event_type": event_type,
            }
            push_service.dispatch_push_async(
                targets,
                push_payload,
                [draft.user_id for draft in drafts],
            )
    return event.id

```

```python
"""Web push endpoints (Phase 10): VAPID public key, subscribe, unsubscribe,
and a test send so the demo can prove delivery end to end."""

import uuid

from fastapi import APIRouter, Depends, HTTPException, status
from pydantic import BaseModel, Field
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.core.db import get_db
from app.core.vapid import public_key_for_browser
from app.dependencies import require_role
from app.models.push import PushSubscription
from app.models.user import User
from app.services.push_service import send_to_user_sync

router = APIRouter(prefix="/push", tags=["push"])


class SubscriptionBody(BaseModel):
    endpoint: str = Field(min_length=1, max_length=1000)
    keys: dict[str, str]
    user_agent: str | None = None


class UnsubscribeBody(BaseModel):
    endpoint: str = Field(min_length=1, max_length=1000)


@router.get("/public-key", summary="VAPID public key for PushManager.subscribe")
def get_public_key(current_user: User = Depends(require_role())) -> dict:
    return {"public_key": public_key_for_browser()}


@router.post("/subscriptions", summary="Register this browser for pushes")
def subscribe(
    body: SubscriptionBody,
    db: Session = Depends(get_db),
    current_user: User = Depends(require_role()),
) -> dict:
    p256dh = body.keys.get("p256dh")
    auth = body.keys.get("auth")
    if not p256dh or not auth:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail="Subscription keys must include p256dh and auth.",
        )

    existing = db.execute(
        select(PushSubscription).where(PushSubscription.endpoint == body.endpoint)
    ).scalar_one_or_none()
    if existing is None:
        existing = PushSubscription(
            id=uuid.uuid4(),
            user_id=current_user.id,
            endpoint=body.endpoint,
            p256dh=p256dh,
            auth=auth,
            user_agent=body.user_agent,
        )
        db.add(existing)
    else:
        # A subscription belongs to whoever holds it now (e.g. re-login).
        existing.user_id = current_user.id
        existing.p256dh = p256dh
        existing.auth = auth
        existing.user_agent = body.user_agent
    db.flush()
    return {"status": "subscribed"}


@router.post("/subscriptions/delete", summary="Remove a browser subscription")
def unsubscribe(
    body: UnsubscribeBody,
    db: Session = Depends(get_db),
    current_user: User = Depends(require_role()),
) -> dict:
    existing = db.execute(
        select(PushSubscription).where(PushSubscription.endpoint == body.endpoint)
    ).scalar_one_or_none()
    if existing is not None and existing.user_id == current_user.id:
        db.delete(existing)
        db.flush()
    return {"status": "unsubscribed"}


@router.post("/test", summary="Send a test push to all my devices")
def test_push(
    db: Session = Depends(get_db),
    current_user: User = Depends(require_role()),
) -> dict:
    result = send_to_user_sync(
        db,
        current_user.id,
        {
            "title": "TrackSkill AI test notification",
            "body": "If you can read this, Web Push delivery works end to end.",
            "link": "/dashboard",
        },
    )
    return result
```

```python
def _scaled(point: tuple[float, float], origin_x: float, origin_y: float, span: float):
    """Map a 32-space point into a span x span box anchored at origin."""
    return (origin_x + point[0] / 32.0 * span, origin_y + point[1] / 32.0 * span)


def draw_mark(draw: ImageDraw.ImageDraw, origin_x: float, origin_y: float, span: float) -> None:
    """Draw the rising milestone path inside the given box."""
    stroke = max(2, int(span * (2.6 / 32.0)))
    radius = stroke / 2
    points = [_scaled(p, origin_x, origin_y, span) for p in PATH_POINTS]

    for start, end in zip(points, points[1:]):
        draw.line([start, end], fill=MARIGOLD, width=stroke)
    for centre in points:
        draw.ellipse(
            [centre[0] - radius, centre[1] - radius, centre[0] + radius, centre[1] + radius],
            fill=MARIGOLD,
        )
    for centre_scaled in DOTS:
        cx, cy = _scaled(centre_scaled, origin_x, origin_y, span)
        dot = span * (2.1 / 32.0)
        draw.ellipse([cx - dot, cy - dot, cx + dot, cy + dot], fill=PAPER)

    for start, end in (
        (_scaled(ARROW_A, origin_x, origin_y, span), _scaled(ARROW_TIP, origin_x, origin_y, span)),
        (_scaled(ARROW_TIP, origin_x, origin_y, span), _scaled(ARROW_B, origin_x, origin_y, span)),
    ):
        draw.line([start, end], fill=MARIGOLD, width=stroke)
        for centre in (start, end):
            draw.ellipse(

                fill=MARIGOLD,
            )
```

```javascript
/*
 * TrackSkill AI service worker.
 *
 * Strategies:
 * - App shell pages (navigations): network-first, every successful HTML
 *   response is cached; offline falls back to the cached page or /offline.
 * - Static assets (/_next/static, icons, splash, manifest): cache-first.
 * - Other same-origin GETs (RSC flights, prefetches): stale-while-revalidate.
 * - API GETs (the FastAPI origin): stale-while-revalidate with a 30-second
 *   freshness window — young entries are served from cache instantly while a
 *   background fetch refreshes them; older entries try the network first so
 *   dashboards stay live, falling back to cache when offline.
 * - Auth endpoints (/auth/*): never cached — login/refresh always hit the
 *   network so credentials can never be served from a stale copy.
 * - Offline follow-up responses: POSTs to /trainee/followups/{id}/respond are
 *   queued in IndexedDB and replayed by a Background Sync event, then the page
 *   is told via postMessage so it can refresh.
 *
 * Bump VERSION whenever the precache list changes; the browser detects SW
 * updates by byte-diff of this file and the page shows "Refresh to update".
 */

const VERSION = "trackskill-v1";
const PAGES_CACHE = `pages-${VERSION}`;

/* ── IndexedDB: the offline follow-up queue ──────────────────────────────── */

const DB_NAME = "trackskill-sw";
const QUEUE_STORE = "queue";
const META_STORE = "meta";

function openDb() {
  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, 1);
    request.onupgradeneeded = () => {
      const db = request.result;
      if (!db.objectStoreNames.contains(QUEUE_STORE)) {
        db.createObjectStore(QUEUE_STORE, { autoIncrement: true });
      }
      if (!db.objectStoreNames.contains(META_STORE)) {
        db.createObjectStore(META_STORE);
      }
    };
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error);
  });
}

async function idbRun(storeName, mode, run) {
  const db = await openDb();
  return new Promise((resolve, reject) => {
    const tx = db.transaction(storeName, mode);
    const store = tx.objectStore(storeName);
    let result;
    try {
      result = run(store);
    } catch (error) {
      reject(error);
      return;
    }
    tx.oncomplete = () => resolve(result && result.result !== undefined ? result.result : result);
    tx.onerror = () => reject(tx.error);
  });
}

function queueAdd(item) {
  return idbRun(QUEUE_STORE, "readwrite", (store) => store.add(item));
}

function queueAll() {
  return idbRun(QUEUE_STORE, "readonly", (store) => store.getAll());
}

function queueAllKeys() {
  return idbRun(QUEUE_STORE, "readonly", (store) => store.getAllKeys());
}

function queueDelete(key) {
  return idbRun(QUEUE_STORE, "readwrite", (store) => store.delete(key));
}

function metaGet(key) {
  return idbRun(META_STORE, "readonly", (store) => store.get(key));
}

function metaSet(key, value) {
  return idbRun(META_STORE, "readwrite", (store) => store.put(value, key));
}

async function putCapped(cacheName, request, response, cap) {
  const cache = await caches.open(cacheName);
  await cache.put(request, response);
  const keys = await cache.keys();
  if (keys.length > cap) {
    for (const staleKey of keys.slice(0, keys.length - cap)) {
      await cache.delete(staleKey);
    }
  }
}

function cacheable(response) {
  return response && response.ok && response.type !== "opaque";
}

function withAge(response) {
  const dateHeader = response.headers.get("date");
  const age = dateHeader ? Date.now() - new Date(dateHeader).getTime() : Infinity;
  return { response, age: Number.isFinite(age) ? age : 0 };
}

 * answered immediately (revalidation continues in the background via
 * event.waitUntil); anything older tries the network first so live data
 * wins, falling back to cache when offline.
 *
 * Returns { respond, revalidate }: call event.respondWith(respond) and
 * event.waitUntil(revalidate) SYNCHRONOUSLY during event dispatch — calling
 * waitUntil after an await inside respondWith rejects the event. */
function staleWhileRevalidate(request, cacheName, cap, freshMs) {
  const revalidate = fetch(request)
    .then(async (response) => {
      if (cacheable(response)) {
        await putCapped(cacheName, request, response.clone(), cap);
      }
      return response;
    })
    .catch(() => null);

  const respond = (async () => {
    const cache = await caches.open(cacheName);
    const cached = withAge(await cache.match(request));
    if (cached.response && cached.age < freshMs) {
      return cached.response;
    }
    const network = await revalidate;
    if (network) {
      return network.clone();
    }
    return cached.response ?? null;
  })();

  return { respond, revalidate };
}

/* ── Install / activate ──────────────────────────────────────────────────── */

self.addEventListener("install", (event) => {
  event.waitUntil(
    (async () => {
      const cache = await caches.open(PAGES_CACHE);
      await Promise.allSettled(
        PRECACHE_URLS.map(async (url) => {
          const response = await fetch(new Request(url, { credentials: "same-origin" }));
          if (response.ok) {
            await cache.put(new Request(url), response);
          }
        }),
      );
    })(),
  );
});

self.addEventListener("activate", (event) => {
  event.waitUntil(
    (async () => {
      for (const key of await caches.keys()) {
        if (!key.endsWith(VERSION)) {
          await caches.delete(key);
        }
      }
      await self.clients.claim();
      await replayQueue().catch(() => undefined);
    })(),
  );
});

/* ── Fetch strategies ────────────────────────────────────────────────────── */

async function handleNavigation(request) {
  try {
    const response = await fetch(request);
    if (cacheable(response)) {
      await putCapped(PAGES_CACHE, request, response.clone(), PAGES_CAP);
    }
    return response;
  } catch {
    const cache = await caches.open(PAGES_CACHE);
    return (
      (await cache.match(request)) ??
      (await cache.match("/dashboard")) ??
      (await cache.match("/offline")) ??
      new Response(
        "<!doctype html><title>Offline</title><p>You are offline.</p>",
        { headers: { "Content-Type": "text/html" }, status: 503 },
      )
    );
  }
}

async function handleFollowupRespond(request) {
  try {
    return await fetch(request);
  } catch {
    const body = await request.clone().arrayBuffer();
    const headers = {};
    request.headers.forEach((value, key) => {
      if (["content-type", "authorization"].includes(key.toLowerCase())) {
        headers[key] = value;
      }
    });
    await queueAdd({
      url: request.url,
      headers,
      body,
      attempts: 0,
      created_at: new Date().toISOString(),
    });
    if (self.registration.sync) {
      try {
        await self.registration.sync.register("tsk-followup-replay");
      } catch {
        // Background Sync unavailable — the queue replays on next activate,
        // next online event or explicit REPLAY_NOW from the page.
      }
    }
    await notifyClients({ type: "followup-queued" });
    return new Response(JSON.stringify({ queued: true, offline: true }), {
      status: 202,
      headers: { "Content-Type": "application/json" },
    });
  }
}

self.addEventListener("fetch", (event) => {
  const { request } = event;
  if (request.method !== "GET" && !(request.method === "POST" && RESPOND_PATTERN.test(new URL(request.url).pathname))) {
    return; // all other mutations go straight to the network
  }

  const url = new URL(request.url);
  const sameOrigin = url.origin === self.location.origin;
  const pathname = url.pathname;

  if (request.method === "POST" && RESPOND_PATTERN.test(pathname)) {
    event.respondWith(handleFollowupRespond(request));
    return;
  }

  if (pathname.startsWith("/auth/")) {
    return; // auth is network-only, never cached and never queued
  }

  if (request.mode === "navigate") {
    event.respondWith(handleNavigation(request));
    return;
  }

  if (!sameOrigin) {
    // Cross-origin API GET — stale-while-revalidate (handles the FastAPI data).
    if (pathname.startsWith("/ws")) {
      return;
    }
    const { respond, revalidate } = staleWhileRevalidate(request, API_CACHE, API_CAP, API_FRESH_MS);
    event.waitUntil(revalidate.then(() => undefined));
    event.respondWith(
      respond.then(
        (response) =>
          response ??
          new Response(JSON.stringify({ detail: "Offline and no cached data." }), {
            status: 503,
            headers: { "Content-Type": "application/json" },
          }),
        () =>
          new Response(JSON.stringify({ detail: "Offline and no cached data." }), {
            status: 503,
            headers: { "Content-Type": "application/json" },
          }),
      ),
    );
    return;
  }

  if (STATIC_PATTERNS.some((pattern) => pattern.test(pathname))) {
    event.respondWith(
      (async () => {
        const cache = await caches.open(STATIC_CACHE);
        const cached = await cache.match(request);
        if (cached) {
          return cached;
        }
        const response = await fetch(request);
        if (cacheable(response)) {
          await cache.put(request, response.clone());
        }
        return response;
      })(),
    );
    return;
  }

  if (RSC_PATTERN.test(pathname) || request.headers.has("Next-Router-Prefetch")) {
    const { respond, revalidate } = staleWhileRevalidate(request, PAGES_CACHE, PAGES_CAP, 0);
    event.waitUntil(revalidate.then(() => undefined));
    event.respondWith(
      respond.then(
        (response) => response ?? new Response("{}", { status: 503, headers: { "Content-Type": "application/json" } }),
        () => new Response("{}", { status: 503, headers: { "Content-Type": "application/json" } }),
      ),
    );
  }
});

/* ── Background sync: replay offline follow-up responses ─────────────────── */

async function notifyClients(message) {
  const clients = await self.clients.matchAll({ includeUncontrolled: true });
  for (const client of clients) {
    client.postMessage(message);
  }
}

async function replayQueue() {
  const items = await queueAll();
  const keys = await queueAllKeys();
  let pendingAuth = false;

  for (let index = 0; index < items.length; index += 1) {
    const item = items[index];
    const key = keys[index];
    try {
      const token = (await metaGet("replay-token")) || null;
      const headers = { ...item.headers };
      if (token) {
        headers.Authorization = `Bearer ${token}`;
      }
      const response = await fetch(item.url, {
        method: "POST",
        headers,
        body: item.body,
      });
      if (response.status === 401) {
        pendingAuth = true;
        await metaSet("replay-token", null);
        continue;
      }
      await queueDelete(key);
      await notifyClients({
        type: "followup-synced",
        url: item.url,
        status: response.status,
      });
    } catch {
      const attempts = (item.attempts ?? 0) + 1;
      await queueDelete(key);
      if (attempts >= MAX_REPLAY_ATTEMPTS) {
        await notifyClients({ type: "followup-failed", url: item.url });
      } else {
        await queueAdd({ ...item, attempts });
      }
    }
  }

  if (pendingAuth) {
    await notifyClients({ type: "followup-auth-needed" });
    if (self.registration.sync) {
      try {
        await self.registration.sync.register("tsk-followup-replay");
      } catch {
        // The page will send REPLAY_NOW once it refreshes the token.
      }
    }
  }
}

self.addEventListener("sync", (event) => {
  if (event.tag === "tsk-followup-replay") {
    event.waitUntil(replayQueue().catch(() => undefined));
  }
});

/* ── Push notifications ──────────────────────────────────────────────────── */

self.addEventListener("push", (event) => {
  let payload = { title: "TrackSkill AI", body: "You have a new update.", link: "/dashboard" };
  try {
    if (event.data) {
      payload = { ...payload, ...event.data.json() };
    }
  } catch {
    if (event.data) {
      payload.body = event.data.text();
    }
  }
  event.waitUntil(
    self.registration.showNotification(payload.title, {
      body: payload.body,
      icon: "/icons/icon-192.png",
      badge: "/icons/icon-192.png",
      tag: payload.event_id || payload.title,
      data: { url: payload.link || "/dashboard" },
    }),
  );
});

self.addEventListener("notificationclick", (event) => {
  event.notification.close();
  const target = event.notification.data?.url || "/dashboard";
  event.waitUntil(
    (async () => {
      const clients = await self.clients.matchAll({ type: "window", includeUncontrolled: true });
      for (const client of clients) {
        if (client.url.includes(self.location.origin)) {
          await client.focus();
          if (client.navigate) {
            await client.navigate(target).catch(() => undefined);
          }
          return;
        }
      }
      await self.clients.openWindow(target);
    })(),
  );
});

/* ── Messages from the page ──────────────────────────────────────────────── */

self.addEventListener("message", (event) => {
  const type = event.data?.type;
  if (type === "SKIP_WAITING") {
    self.skipWaiting();
  } else if (type === "REPLAY_NOW") {
    event.waitUntil(replayQueue().catch(() => undefined));
  } else if (type === "QUEUE_TOKEN") {
    event.waitUntil(
      (async () => {
        await metaSet("replay-token", event.data.token);
        await replayQueue().catch(() => undefined);
      })(),
    );
  }
});
```

```typescript
"use client";

/** Single mount point for all PWA UX: service-worker registration with the
 * "Refresh to update" toast, the custom install prompt (with an iOS Safari
 * hint), "App installed" detection, the push opt-in shown only after login,
 * and offline follow-up sync messages from the service worker. */

import { useQueryClient } from "@tanstack/react-query";
import { Bell, CheckCheck, Download, RefreshCw, X } from "lucide-react";
import { useCallback, useEffect, useState } from "react";

import { ensureFreshToken, getAccessToken } from "@/lib/api";
import {
  enablePush,
  isIosSafari,
  isStandalone,
  pushPermission,
  pushSupported,
  registerServiceWorker,
  requestQueueReplay,
} from "@/lib/push-client";
import { useAuthStore } from "@/stores/auth-store";

type Banner =
  | { kind: "none" }
  | { kind: "update" }
  | { kind: "install" }
  | { kind: "ios-install" }
  | { kind: "push" }
  | { kind: "message"; text: string };

type InstallPromptEvent = Event & { prompt: () => Promise<void> };

const INSTALL_DISMISS_KEY = "pwa-install-dismissed";
const PUSH_CHOICE_KEY = "pwa-push-choice";

export function PwaRoot() {
  const queryClient = useQueryClient();
  const status = useAuthStore((state) => state.status);
  const [banner, setBanner] = useState<Banner>({ kind: "none" });
  const [waitingWorker, setWaitingWorker] = useState<ServiceWorker | null>(null);
  const [promptEvent, setPromptEvent] = useState<InstallPromptEvent | null>(null);
  const [busy, setBusy] = useState(false);

  /* Service worker registration + update detection. */
  useEffect(() => {
    void registerServiceWorker().then((registration) => {
      registration?.addEventListener("updatefound", () => {
        const worker = registration.installing;
        worker?.addEventListener("statechange", () => {
          if (worker.state === "installed" && navigator.serviceWorker.controller) {
            setWaitingWorker(worker);
          }
        });
      });
      if (registration?.waiting && navigator.serviceWorker.controller) {
        setWaitingWorker(registration.waiting);
      }
    });
  }, []);

  /* Update toast. */
  useEffect(() => {
    if (waitingWorker) {
      setBanner({ kind: "update" });
    }
  }, [waitingWorker]);

  /* Install prompt capture (Chromium) / hint (iOS Safari) + installed event. */
  useEffect(() => {
    if (localStorage.getItem(INSTALL_DISMISS_KEY) === "1" || isStandalone()) {
      return;
    }
    const onBeforeInstall = (event: Event) => {
      event.preventDefault();
      setPromptEvent(event as InstallPromptEvent);
      setBanner({ kind: "install" });
    };
    const onInstalled = () => {
      setPromptEvent(null);
      setBanner({ kind: "message", text: "TrackSkill AI installed — find it in your app list." });
    };
    window.addEventListener("beforeinstallprompt", onBeforeInstall);
    window.addEventListener("appinstalled", onInstalled);
    // iOS Safari never fires beforeinstallprompt — offer the manual route.
    const iosTimer = isIosSafari()
      ? window.setTimeout(() => setBanner({ kind: "ios-install" }), 2000)
      : undefined;
    return () => {
      window.removeEventListener("beforeinstallprompt", onBeforeInstall);
      window.removeEventListener("appinstalled", onInstalled);
      if (iosTimer) {
        window.clearTimeout(iosTimer);
      }
    };
  }, []);

  /* Push opt-in — only offered once signed in, never on first load. */
  useEffect(() => {
    if (status !== "authenticated") {
      return;
    }
    if (localStorage.getItem(PUSH_CHOICE_KEY) || !pushSupported()) {
      return;
    }
    if (pushPermission() !== "default") {
      return;
    }
    const timer = window.setTimeout(
      () => setBanner((current) => (current.kind === "none" ? { kind: "push" } : current)),
      1500,
    );
    return () => window.clearTimeout(timer);
  }, [status]);

  /* Offline follow-up queue messages from the service worker. */
  useEffect(() => {
    const onMessage = (event: MessageEvent) => {
      const data = event.data as { type?: string; status?: number } | null;
      if (!data?.type) {
        return;
      }
      if (data.type === "followup-synced") {
        void queryClient.invalidateQueries();
        setBanner({ kind: "message", text: "Offline response synced with the server." });
      } else if (data.type === "followup-queued") {
        setBanner({
          kind: "message",
          text: "You are offline — the response will send automatically once you reconnect.",
        });
      } else if (data.type === "followup-failed") {
        setBanner({
          kind: "message",
          text: "An offline response could not be delivered after several attempts.",
        });
      } else if (data.type === "followup-auth-needed") {
        void ensureFreshToken().then((fresh) => {
          if (!fresh) {
            return;
          }
          const token = getAccessToken();
          void navigator.serviceWorker.ready.then((registration) => {
            registration.active?.postMessage({ type: "QUEUE_TOKEN", token });
          });
        });
      }
    };
    navigator.serviceWorker?.addEventListener("message", onMessage);
    const onOnline = () => requestQueueReplay();
    window.addEventListener("online", onOnline);
    return () => {
      navigator.serviceWorker?.removeEventListener("message", onMessage);
      window.removeEventListener("online", onOnline);
    };
  }, [queryClient]);

  const dismiss = useCallback((storageKey?: string) => {
    if (storageKey) {
      localStorage.setItem(storageKey, "1");
    }
    setBanner({ kind: "none" });
  }, []);

  const applyUpdate = () => {
    waitingWorker?.postMessage({ type: "SKIP_WAITING" });
    navigator.serviceWorker.addEventListener(
      "controllerchange",
      () => window.location.reload(),
      { once: true },
    );
  };

  const runInstall = async () => {
    if (!promptEvent) {
      return;
    }
    setBusy(true);
    try {
      await promptEvent.prompt();
    } finally {
      setBusy(false);
      setPromptEvent(null);
      setBanner({ kind: "none" });
    }
  };

  const turnOnPush = async () => {
    setBusy(true);
    const result = await enablePush();
    localStorage.setItem(PUSH_CHOICE_KEY, result);
    setBusy(false);
    setBanner({
      kind: "message",
      text:
        result === "granted"
          ? "Notifications are on — tracers, verifications and fund decisions will reach your device."
          : result === "denied"
            ? "Notifications are blocked in the browser — change it in site settings."
            : "Notifications are not available on this device.",
    });
  };

  if (banner.kind === "none") {
    return null;
  }

  return (
    <div
      className="fixed inset-x-0 bottom-0 z-[60] flex justify-center px-3 pb-[calc(env(safe-area-inset-bottom)+0.75rem)] sm:justify-end sm:px-6"
      role="status"
      aria-live="polite"
    >
      <div className="w-full max-w-sm rounded-xl border border-border bg-card p-4 shadow-lg">
        {banner.kind === "update" && (
          <CardRow
            icon={<RefreshCw className="h-5 w-5 text-primary" aria-hidden />}
            title="A new version is ready"
            body="Refresh to update TrackSkill AI."
            actionLabel="Refresh to update"
            onAction={applyUpdate}
            onDismiss={() => setBanner({ kind: "none" })}
```

```typescript
"use client";

import { useQueryClient } from "@tanstack/react-query";
import { Loader2 } from "lucide-react";
import { useCallback, useEffect, useRef, useState } from "react";

const TRIGGER_PX = 64;
const MAX_PULL_PX = 96;

/** Mobile pull-to-refresh for touch devices: pulling down at the top of the
 * page refreshes every dashboard query, with a haptic tick where supported.
 * Renders its children plus the fixed drag indicator. */
export function PullToRefresh({ children }: { children: React.ReactNode }) {
  const queryClient = useQueryClient();
  const [pull, setPull] = useState(0);
  const [refreshing, setRefreshing] = useState(false);
  const startY = useRef<number | null>(null);
  const startX = useRef<number>(0);
  const pulling = useRef(false);

  const refresh = useCallback(async () => {
    setRefreshing(true);
    try {
      await queryClient.invalidateQueries();
      // Give refetches a beat to land so the user sees new content.
      await new Promise((resolve) => setTimeout(resolve, 600));
    } finally {
      setRefreshing(false);
    }
  }, [queryClient]);

  useEffect(() => {
    const onTouchStart = (event: TouchEvent) => {
      if (event.touches.length !== 1) {
        return;
      }
      startY.current = event.touches[0].clientY;
      startX.current = event.touches[0].clientX;
      pulling.current = window.scrollY <= 0;
    };

    const onTouchMove = (event: TouchEvent) => {
      if (!pulling.current || startY.current === null || refreshing) {
        return;
      }
      const deltaY = event.touches[0].clientY - startY.current;
      const deltaX = event.touches[0].clientX - startX.current;
      // Only react to deliberate vertical drags.
      if (deltaY <= 0 || Math.abs(deltaX) > Math.abs(deltaY)) {
        setPull(0);
        return;
      }
      if (window.scrollY > 4) {
        pulling.current = false;
        setPull(0);
        return;
      }
      // Rubber-band resistance past the trigger point.
      const distance = deltaY > TRIGGER_PX ? TRIGGER_PX + (deltaY - TRIGGER_PX) * 0.25 : deltaY;
      setPull(Math.min(distance, MAX_PULL_PX));
    };

    const onTouchEnd = async () => {
      if (!pulling.current) {
        return;
      }
      startY.current = null;
      pulling.current = false;
      setPull((current) => {
        if (current >= TRIGGER_PX * 0.8) {
          navigator.vibrate?.(10);
          void refresh();
          return 0;
        }
        return 0;
      });
    };

    document.addEventListener("touchstart", onTouchStart, { passive: true });
    document.addEventListener("touchmove", onTouchMove, { passive: true });
    document.addEventListener("touchend", onTouchEnd, { passive: true });
    return () => {
      document.removeEventListener("touchstart", onTouchStart);
      document.removeEventListener("touchmove", onTouchMove);
      document.removeEventListener("touchend", onTouchEnd);
    };
  }, [refresh, refreshing]);

  const progress = refreshing || pull > 0;

  return (
    <>
      <div
        aria-hidden={!progress}
        className="pointer-events-none fixed inset-x-0 top-0 z-50 flex justify-center transition-opacity"
        style={{
          opacity: progress ? 1 : 0,
          transform: refreshing ? "translateY(12px)" : `translateY(${pull * 0.4}px)`,
        }}
      >
        <span className="mt-2 flex items-center gap-2 rounded-full border border-border bg-card px-3 py-1.5 text-xs text-muted-foreground shadow">
          <Loader2 className={`h-3.5 w-3.5 ${refreshing ? "animate-spin" : ""}`} aria-hidden />
          {refreshing ? "Refreshing…" : pull >= TRIGGER_PX * 0.8 ? "Release to refresh" : "Pull to refresh"}
        </span>
      </div>
      {children}
    </>
  );
}
```

```typescript
function urlBase64ToUint8Array(base64String: string): Uint8Array {
  const padding = "=".repeat((4 - (base64String.length % 4)) % 4);
  const base64 = (base64String + padding).replace(/-/g, "+").replace(/_/g, "/");
  const raw = window.atob(base64);
  const output = new Uint8Array(raw.length);
  for (let index = 0; index < raw.length; index += 1) {
    output[index] = raw.charCodeAt(index);
  }
  return output;
}

/** Ask permission, subscribe to push and register the subscription with the
 * API. Returns "granted" | "denied" | "unsupported" | "error". */
export async function enablePush(): Promise<"granted" | "denied" | "unsupported" | "error"> {
  if (!pushSupported() || !getAccessToken()) {
    return "unsupported";
  }
  try {
    const permission = await Notification.requestPermission();
    if (permission !== "granted") {
      return "denied";
    }
    const registration = await navigator.serviceWorker.ready;
    const vapidPublicKey = await fetchPushPublicKey();
    const existing = await registration.pushManager.getSubscription();
    const subscription =
      existing ??
      (await registration.pushManager.subscribe({
        userVisibleOnly: true,
        applicationServerKey: urlBase64ToUint8Array(vapidPublicKey) as unknown as BufferSource,
      }));
    const json = subscription.toJSON() as {
      endpoint: string;
      keys?: { p256dh?: string; auth?: string };
    };
    if (!json.keys?.p256dh || !json.keys?.auth) {
      return "error";
    }
    await savePushSubscription({
      endpoint: json.endpoint,
      keys: { p256dh: json.keys.p256dh, auth: json.keys.auth },
      user_agent: window.navigator.userAgent,
    });
    return "granted";
  } catch {
    return "error";
  }
}

/** Remove the browser subscription and tell the API to drop it. */
export async function disablePush(): Promise<void> {
  if (!pushSupported()) {
    return;
  }
  try {
    const registration = await navigator.serviceWorker.ready;
    const subscription = await registration.pushManager.getSubscription();
    if (subscription) {
      const json = subscription.toJSON() as { endpoint?: string };
      await subscription.unsubscribe();
      if (json.endpoint) {
        await deletePushSubscription(json.endpoint);
      }
    }
  } catch {
    // best-effort cleanup
  }
}

export function pushPermission(): NotificationPermission | "unsupported" {
  if (typeof window === "undefined" || !("Notification" in window)) {
    return "unsupported";
  }
  return Notification.permission;
}
```

```typescript
 *

  authExpiredListeners.add(listener);

// Endpoints whose own 401 responses must never trigger a refresh round-trip.
const AUTH_PATHS = ["/auth/login", "/auth/register", "/auth/refresh", "/auth/logout"];

let refreshInFlight: Promise<boolean> | null = null;

async function requestRefresh(): Promise<boolean> {
  try {
    const response = await fetch(`${API_URL}/auth/refresh`, {
      method: "POST",
      credentials: "include",
      cache: "no-store",
    });
    if (!response.ok) {
      return false;
    }
    const data = (await response.json()) as { access_token: string };
    accessToken = data.access_token;
    return true;
  } catch {
    return false;
  }
}

function refreshAccessToken(): Promise<boolean> {
  if (!refreshInFlight) {
    refreshInFlight = requestRefresh().finally(() => {
      refreshInFlight = null;
    });
  }
  return refreshInFlight;
}

interface InternalOptions extends RequestOptions {
  method?: string;
  body?: unknown;
  isRetry?: boolean;
}

async function request<T>(path: string, options: InternalOptions = {}): Promise<T> {
  const { method = "GET", body, headers, signal, isRetry = false } = options;

  const response = await fetch(`${API_URL}${path}`, {
    method,
    headers: {
      ...(body !== undefined ? { "Content-Type": "application/json" } : {}),
      ...(accessToken ? { Authorization: `Bearer ${accessToken}` } : {}),
      ...headers,
    },
    body: body !== undefined ? JSON.stringify(body) : undefined,
    credentials: "include",
    cache: "no-store",
    signal,
  });

  if (
    response.status === 401 &&
    !isRetry &&
    !AUTH_PATHS.some((authPath) => path.startsWith(authPath))
  ) {
    const refreshed = await refreshAccessToken();
    if (refreshed) {
      return request<T>(path, { ...options, isRetry: true });
    }
    accessToken = null;
    authExpiredListeners.forEach((listener) => listener());
  }

  if (!response.ok) {
    let message = `Request failed with status ${response.status}`;
    let details: unknown = null;
    try {
      const problem = await response.json();
      details = problem;
      if (typeof problem?.detail === "string") {
        message = problem.detail;
      }
    } catch {
      // The response had no JSON body; keep the generic message.
    }
    throw new ApiError(response.status, message, details);
  }

  if (response.status === 204) {
    return undefined as T;
  }
  return (await response.json()) as T;
}

export const api = {
  get: <T>(path: string, options: RequestOptions = {}) =>

  post: <T>(path: string, body?: unknown, options: RequestOptions = {}) =>
    request<T>(path, { ...options, method: "POST", body }),
  patch: <T>(path: string, body?: unknown, options: RequestOptions = {}) =>
    request<T>(path, { ...options, method: "PATCH", body }),
  delete: <T>(path: string, options: RequestOptions = {}) =>
    request<T>(path, { ...options, method: "DELETE" }),
};

 *
 * NEXT_PUBLIC_WS_URL may be the full endpoint (ws://host/ws) or just the
 * origin (ws://host) — in the latter case /ws is appended. When unset, the
 * URL is derived from the API base URL.

  if (!/\/ws$/.test(base)) {

}
```

```typescript
import { useQueryClient } from "@tanstack/react-query";
import { useRouter } from "next/navigation";
import { useEffect } from "react";

import { PullToRefresh } from "@/components/pwa/pull-to-refresh";
import { useRealtimeConnection } from "@/hooks/use-realtime-connection";
import { useAuthStore } from "@/stores/auth-store";
import { useRealtimeStore } from "@/stores/realtime-store";

/** Authenticated shell wrapper: guards unauthenticated visitors, opens the
 * realtime socket and invalidates React Query caches when domain events
 * arrive. The visual chrome (sidebar, top bar, tabs) is rendered per role
 * page via DashboardShell. */
export default function DashboardLayout({ children }: { children: React.ReactNode }) {
  const router = useRouter();
  const queryClient = useQueryClient();
  const user = useAuthStore((state) => state.user);
  const status = useAuthStore((state) => state.status);
  const bootstrap = useAuthStore((state) => state.bootstrap);

  useRealtimeConnection(status === "authenticated");

  useEffect(() => {
    if (status === "idle") {
      void bootstrap();
    }
  }, [status, bootstrap]);

  useEffect(() => {
    if (status === "unauthenticated") {
      router.replace("/login");
    }
  }, [status, router]);

  // Any domain event may change data anywhere — invalidate all queries.
  useEffect(() => {
    const unsubscribe = useRealtimeStore.subscribe((state) => {
      if (state.lastEvent) {
        void queryClient.invalidateQueries();
      }
    });
    return unsubscribe;
  }, [queryClient]);

  if (status !== "authenticated" || !user) {
    return (
      <div className="flex min-h-dvh items-center justify-center bg-background text-muted-foreground">
        <p role="status">
          {status === "loading" ? "Loading your workspace…" : "Redirecting to sign-in…"}
        </p>
      </div>
    );
  }

  return (
    <PullToRefresh>
      {children}
    </PullToRefresh>
  );
}
```

```javascript
 *
 * Wraps the running TrackSkill AI web app in a native window. Point it at a
 * local `next start` or a deployed origin:
 *
 *   TRACKSKILL_URL=http://localhost:3000 npx electron electron/main.cjs

 *   # then to produce an installer:

 */

const { app, BrowserWindow, shell } = require("node:electron");

const TARGET = process.env.TRACKSKILL_URL || "http://localhost:3000";

function createWindow() {

    minWidth: 380,

      nodeIntegration: false,
      sandbox: true,
    },
  });

  win.webContents.setWindowOpenHandler(({ url }) => {

  app.on("activate", () => {
    if (BrowserWindow.getAllWindows().length === 0) {
```
